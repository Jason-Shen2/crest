# Agent Change Review Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first production path for agent code-change review: edit tools emit `ChangeOperation`, the renderer derives `ChangeSet`, the agent/system path supplies `ChangeOutline`, and the UI renders a validated `ChangeReview`.

**Architecture:** Edit tools attach structured change metadata to tool result details. The frontend derives and validates review data from the run transcript so the current `AgentProgressView` can show module -> file -> code-block review without raw tool cards. `ChangeOutline` generation is isolated behind a model-call seam; `ChangeReview` is always generated by deterministic merge logic.

**Tech Stack:** TypeScript, React, Vitest, Electron main/renderer IPC, existing agent tool result details, existing `AgentProgressView`, existing preview framework.

Add helpers in the same file.

```ts
function operationFromMessage(message: PiAgentMessage): ChangeOperation | undefined {
    if (message.role !== "toolResult" || message.isError) return undefined;
    const details = message.details;
    if (!details || typeof details !== "object") return undefined;
    const op = (details as { changeOperation?: unknown }).changeOperation;
    if (!op || typeof op !== "object") return undefined;
    const record = op as Partial<ChangeOperation>;
    if (!record.id || !record.path || !record.kind) return undefined;
    return record as ChangeOperation;
}

export function parseUnifiedPatchHunks(filePath: string, patch: string): ChangeSetHunk[] {
    const hunks: ChangeSetHunk[] = [];
    const parts = patch.split(/(?=^@@ )/m).filter((part) => part.startsWith("@@ "));
    for (let index = 0; index < parts.length; index++) {
        const part = parts[index];
        const header = part.split("\n")[0] ?? "";
        const match = header.match(/^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@/);
        if (!match) continue;
        const additions = part.split("\n").filter((line) => line.startsWith("+") && !line.startsWith("+++")).length;
        const deletions = part.split("\n").filter((line) => line.startsWith("-") && !line.startsWith("---")).length;
        hunks.push({
            id: `${filePath}:hunk-${index + 1}`,
            filePath,
            oldStart: Number(match[1]),
            oldLines: Number(match[2] ?? "1"),
            newStart: Number(match[3]),
            newLines: Number(match[4] ?? "1"),
            additions,
            deletions,
            diffPreview: part.trim(),
        });
    }
    return hunks;
}
```

Add builder functions.

```ts
export function buildChangeSet(runId: string, operations: ChangeOperation[]): ChangeSet | undefined {
    if (operations.length === 0) return undefined;
    const files = new Map<string, ChangeSetFile>();
    for (const operation of operations) {
        const existing = files.get(operation.path);
        const hunks = operation.patch ? parseUnifiedPatchHunks(operation.path, operation.patch) : [];
        const additions = hunks.reduce((sum, hunk) => sum + hunk.additions, 0);
        const deletions = hunks.reduce((sum, hunk) => sum + hunk.deletions, 0);
        files.set(operation.path, {
            path: operation.path,
            previousPath: operation.previousPath ?? existing?.previousPath,
            status: operation.kind === "create" ? "added" : operation.kind === "delete" ? "deleted" : operation.kind === "rename" ? "renamed" : "modified",
            additions: (existing?.additions ?? 0) + additions,
            deletions: (existing?.deletions ?? 0) + deletions,
            hunks: [...(existing?.hunks ?? []), ...hunks],
        });
    }
    return { id: runId, runId, source: "change-operations", files: [...files.values()] };
}

export function deriveAgentChangeReview(run: PiRun, outline?: ChangeOutline): ChangeReview | undefined {
    const changeSet = buildChangeSet(run.runId, extractChangeOperations(run));
    return changeSet ? buildChangeReview(changeSet, outline) : undefined;
}
```

Add merge functions.

```ts
export function buildChangeReview(changeSet: ChangeSet, outline?: ChangeOutline): ChangeReview {
    if (!outline || outline.changeSetId !== changeSet.id) {
        return fallbackChangeReview(changeSet, outline ? [{ kind: "stale-outline", message: "Outline does not match ChangeSet." }] : []);
    }
    const filesByPath = new Map(changeSet.files.map((file) => [file.path, file]));
    const used = new Set<string>();
    const warnings: ChangeReviewWarning[] = [];
    const modules: ChangeReviewModule[] = [];
    for (const outlineModule of outline.modules) {
        const files: ChangeReviewFile[] = [];
        for (const outlineFile of outlineModule.files) {
            const file = filesByPath.get(outlineFile.path);
            if (!file) {
                warnings.push({ kind: "unknown-file", message: `Outline referenced unknown file: ${outlineFile.path}` });
                continue;
            }
            used.add(file.path);
            files.push({ ...file, summary: outlineFile.summary, hunkGroups: buildHunkGroups(file, outlineFile, warnings) });
        }
        if (files.length === 0) {
            warnings.push({ kind: "empty-module", message: `Outline module has no valid files: ${outlineModule.title}` });
            continue;
        }
        modules.push(makeReviewModule(outlineModule.id ?? `module-${modules.length + 1}`, outlineModule.title, outlineModule.summary, files));
    }
    if (modules.length === 0) return fallbackChangeReview(changeSet, warnings);
    return {
        changeSetId: changeSet.id,
        changeSet,
        modules,
        ungroupedFiles: changeSet.files.filter((file) => !used.has(file.path)).map((file) => ({ ...file, hunkGroups: [] })),
        warnings,
    };
}
```

Add hunk helper and fallback.

```ts
function buildHunkGroups(file: ChangeSetFile, outlineFile: ChangeOutlineFile, warnings: ChangeReviewWarning[]): ChangeReviewHunkGroup[] {
    const hunksById = new Map(file.hunks.map((hunk) => [hunk.id, hunk]));
    return (outlineFile.hunkGroups ?? []).flatMap((group, index) => {
        const validHunks = group.hunkIds.map((id) => hunksById.get(id)).filter((hunk): hunk is ChangeSetHunk => hunk != null);
        if (validHunks.length === 0) {
            warnings.push({ kind: "unknown-hunk", message: `Outline referenced unknown hunk in ${file.path}.` });
            return [];
        }
        return [{
            id: `${file.path}:group-${index + 1}`,
            title: group.title,
            summary: group.summary,
            impact: group.impact,
            hunkIds: validHunks.map((hunk) => hunk.id),
            additions: validHunks.reduce((sum, hunk) => sum + hunk.additions, 0),
            deletions: validHunks.reduce((sum, hunk) => sum + hunk.deletions, 0),
        }];
    });
}

function makeReviewModule(id: string, title: string, summary: string, files: ChangeReviewFile[]): ChangeReviewModule {
    return {
        id,
        title,
        summary,
        files,
        additions: files.reduce((sum, file) => sum + file.additions, 0),
        deletions: files.reduce((sum, file) => sum + file.deletions, 0),
    };
}

function fallbackChangeReview(changeSet: ChangeSet, warnings: ChangeReviewWarning[]): ChangeReview {
    const files = changeSet.files.map((file) => ({ ...file, hunkGroups: [] }));
    return {
        changeSetId: changeSet.id,
        changeSet,
        modules: [makeReviewModule("code-changes", "Code changes", "Reviewed real files changed by this agent run.", files)],
        ungroupedFiles: [],
        warnings: warnings.length > 0 ? warnings : [{ kind: "missing-outline", message: "No valid change outline was available." }],
    };
}
```

```ts
export interface ChangeOutline {
    changeSetId: string;
    modules: Array<{
        id?: string;
        title: string;
        summary: string;
        files: Array<{
            path: string;
            summary?: string;
            hunkGroups?: Array<{
                title: string;
                summary: string;
                impact?: string;
                hunkIds: string[];
            }>;
        }>;
    }>;
}

export function parseChangeOutlineText(text: string): ChangeOutline | undefined {
    try {
        const parsed = JSON.parse(text) as ChangeOutline;
        if (!parsed || typeof parsed.changeSetId !== "string" || !Array.isArray(parsed.modules)) return undefined;
        return parsed;
    } catch {
        return undefined;
    }
}

export function buildChangeOutlinePrompt(input: {
    userTask: string;
    changeSetJson: string;
    validationSummary?: string;
}): string {
    return [
        "Create a ChangeOutline JSON object for reviewing agent code changes.",
        "Use only file paths and hunk IDs present in the supplied ChangeSet.",
        "Do not invent files, diff stats, line numbers, or hunk IDs.",
        "Return JSON only.",
        "",
        `User task:\n${input.userTask}`,
        "",
        `ChangeSet:\n${input.changeSetJson}`,
        "",
        input.validationSummary ? `Validation:\n${input.validationSummary}` : "",
    ].filter(Boolean).join("\n");
}
```

```ts
            signal: input.signal,

    return parseChangeOutlineText(text);
```

Add compact ChangeSet helpers so the model call can stay in main without importing renderer code.

```ts
import type { AgentMessage } from "../types";

export function extractChangeOperationsFromMessages(messages: AgentMessage[]): unknown[] {
    return messages.flatMap((message) => {
        if ((message as { role?: string }).role !== "toolResult") return [];
        if ((message as { isError?: boolean }).isError) return [];
        const details = (message as { details?: unknown }).details;
        if (!details || typeof details !== "object") return [];
        const operation = (details as { changeOperation?: unknown }).changeOperation;
        return operation && typeof operation === "object" ? [operation] : [];
    });
}

export function buildCompactChangeSetJson(runId: string, operations: unknown[]): string | undefined {
    if (operations.length === 0) return undefined;
    return JSON.stringify({
        id: runId,
        runId,
        source: "change-operations",
        operations: operations.map((operation) => {
            const record = operation as { id?: string; kind?: string; path?: string; patch?: string };
            const hunkHeaders = typeof record.patch === "string" ? record.patch.match(/^@@ .+ @@/gm) ?? [] : [];
            return {
                id: record.id,
                kind: record.kind,
                path: record.path,
                hunkHeaders,
                patchPreview: typeof record.patch === "string" ? record.patch.slice(0, 1200) : undefined,
            };
        }),
    });
}
```

```ts
private async applyRunFinishedHook(run: AgentRun): Promise<void> {
    if (!this.onRunFinished) return;
    const patch = await this.onRunFinished(run).catch((err) => {
        console.error(`[pane-session] run finished hook failed for ${run.runId}:`, err);
        return undefined;
    });
    if (!patch) return;
    const current = this.runs.find((item) => item.runId === run.runId);
    if (!current) return;
    this.setRun({ ...current, ...patch });
    for (const listener of this.listeners) {
        listener({ type: "snapshot", messages: this.messages, runs: this.runs, status: this.status } as any);
    }
}
```
